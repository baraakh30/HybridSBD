\begin{algorithm}
\caption{Threshold-Based Shot Boundary Detection with Constraints}
\begin{algorithmic}[1]
\Require Correlation array $C$, threshold $\theta$, minimum scene length $L_{min}$, maximum dissolve window $W_{max}$
\Ensure List of boundary frame indices $B$

\State $I_{below} \gets \{i \mid C[i] < \theta\}$ \Comment{Find frames below threshold}
\If{$I_{below} = \emptyset$}
    \State \Return $\emptyset$
\EndIf

\State \textbf{Step 1: Group consecutive below-threshold frames}
\State $groups \gets []$
\State $current\_group \gets [I_{below}[0]]$
\For{$i = 1$ to $|I_{below}| - 1$}
    \If{$I_{below}[i] = I_{below}[i-1] + 1$}
        \State $current\_group.append(I_{below}[i])$
    \Else
        \State $groups.append(current\_group)$
        \State $current\_group \gets [I_{below}[i]]$
    \EndIf
\EndFor
\State $groups.append(current\_group)$

\State \textbf{Step 2: Filter groups and identify boundaries}
\State $boundaries \gets []$
\For{each $group$ in $groups$}
    \If{$|group| \leq W_{max}$}
        \State $boundaries.append(group[0])$ \Comment{Single boundary or small dissolve}
    \Else
        \State $boundaries.append(group[0])$ \Comment{Dissolve start}
        \State $boundaries.append(group[-1])$ \Comment{Dissolve end}
    \EndIf
\EndFor

\State \textbf{Step 3: Enforce minimum scene length}
\If{$boundaries = \emptyset$}
    \State \Return $\emptyset$
\EndIf
\State $B \gets [boundaries[0]]$
\For{each $b$ in $boundaries[1:]$}
    \If{$b - B[-1] \geq L_{min}$}
        \State $B.append(b)$
    \EndIf
\EndFor

\State \Return $B$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}
\caption{Simple Threshold-Based Boundary Detection}
\begin{algorithmic}[1]
\Require Correlation array $C$, threshold $\theta$
\Ensure List of boundary frame indices $B$

\State $B \gets []$
\For{$i = 0$ to $|C| - 1$}
    \If{$C[i] < \theta$}
        \State $B.append(i)$
    \EndIf
\EndFor
\State \Return $B$
\end{algorithmic}
\end{algorithm}
